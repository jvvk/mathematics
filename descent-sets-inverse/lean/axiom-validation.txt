def Stanley.beta : ℕ → Finset ℕ → Finset ℕ → ℕ :=
fun n S T => {w | Stanley.descP w = S ∧ Stanley.descP w⁻¹ = T}.card
def Stanley.Alt.g : ℕ → ℕ :=
fun n => {w | Stanley.descP w = Stanley.Alt.altS n ∧ Stanley.descP w⁻¹ = Stanley.Alt.altS n}.card
def Stanley.Alt.E : ℕ → ℕ :=
fun n => (Stanley.Alt.row n).getD n 0
