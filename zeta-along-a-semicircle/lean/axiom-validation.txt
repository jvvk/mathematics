ZetaArc.re_ratio : ∀ (r u : ℝ),
  r ≠ 0 →
    Real.sin (u / 2) ≠ 0 →
      (-↑r * Complex.I * Complex.exp (↑u * Complex.I) / (1 - ↑r * Complex.exp (↑u * Complex.I) - (1 - ↑r))).re =
        Real.cos (u / 2) / (2 * Real.sin (u / 2))
