def TwoTri.Inv.{u_1} : {K : Type u_1} →
  [Field K] → [LinearOrder K] → Finset (K × K) → Finset (Sym2 (K × K)) → Finset (Sym2 (K × K)) → Prop :=
fun {K} [Field K] [LinearOrder K] P A B =>
  ∃ x y z b₁ b₂ b₃ t₁ t₂ t₃,
    TwoTri.Face P A x y z ∧
      TwoTri.Face P B b₁ b₂ b₃ ∧
        TwoTri.Face P B t₁ t₂ t₃ ∧
          (∀ (v : K × K), v = x ∨ v = y ∨ v = z → v ≠ b₁ ∧ v ≠ b₂ ∧ v ≠ b₃) ∧
            (t₁ ≠ x ∧ t₁ ≠ y) ∧
              (t₂ ≠ x ∧ t₂ ≠ y) ∧
                (t₃ ≠ x ∧ t₃ ≠ y) ∧
                  (∃ w, (w = t₁ ∨ w = t₂ ∨ w = t₃) ∧ w ≠ b₁ ∧ w ≠ b₂ ∧ w ≠ b₃) ∧
                    (∃ s, 0 < s ∧ s < 1 ∧ TwoTri.InTri b₁ b₂ b₃ (TwoTri.lerp x y s)) ∧
                      ∃ s, 0 < s ∧ s < 1 ∧ TwoTri.InTri t₁ t₂ t₃ (TwoTri.lerp x y s)
