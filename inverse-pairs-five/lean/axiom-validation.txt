InversePairs.main_ineq : ∀ (p : ℕ) [hp : Fact (Nat.Prime p)],
  InversePairs.WeilBound p →
    ∀ (h L Y : ℕ),
      1 ≤ h →
        p ≤ 1 + L * h →
          1 ≤ Y →
            ∀ (D ε : ℝ),
              0 ≤ ε →
                0 ≤ D →
                  (∀ (n : ℕ), n ≠ 0 → ↑n.divisors.card ≤ D * ↑n ^ ε) →
                    |InversePairs.S p - InversePairs.C p - 1| ≤
                      (↑L * InversePairs.E1 p + ↑h + 1) * InversePairs.w Y +
                          √↑Y / ↑p * (D * (1 + ↑Y) ^ ε + 1 + Real.log ↑p) +
                        1 / ↑p
